   ```python
   """Půdorys jako polygon s typovanými hranami (okap/štít). Čistý, bez UI."""
   from __future__ import annotations

   import math
   from dataclasses import dataclass
   from enum import Enum

   Point = tuple[float, float]


   class EdgeType(str, Enum):
       EAVE = "eave"
       GABLE = "gable"


   @dataclass(frozen=True)
   class Edge:
       type: EdgeType


   def _signed_area(pts: list[Point]) -> float:
       n = len(pts)
       return 0.5 * sum(pts[i][0] * pts[(i + 1) % n][1] - pts[(i + 1) % n][0] * pts[i][1]
                        for i in range(n))


   def _seg_intersect(a, b, c, d) -> bool:
       def orient(p, q, r):
           v = (q[1] - p[1]) * (r[0] - q[0]) - (q[0] - p[0]) * (r[1] - q[1])
           return 0 if abs(v) < 1e-12 else (1 if v > 0 else -1)
       o1, o2, o3, o4 = orient(a, b, c), orient(a, b, d), orient(c, d, a), orient(c, d, b)
       return o1 != o2 and o3 != o4


   def _is_simple(pts: list[Point]) -> bool:
       n = len(pts)
       for i in range(n):
           for j in range(i + 1, n):
               if j == i or (j + 1) % n == i or (i + 1) % n == j:
                   continue
               if _seg_intersect(pts[i], pts[(i + 1) % n], pts[j], pts[(j + 1) % n]):
                   return False
       return True


   @dataclass(frozen=True)
   class Footprint:
       vertices: tuple[Point, ...]
       edges: tuple[Edge, ...]

       @classmethod
       def from_lists(cls, vertices: list[Point], edge_types: list[str]) -> "Footprint":
           pts = [(float(x), float(y)) for x, y in vertices]
           if len(pts) < 3:
               raise ValueError("Půdorys musí mít alespoň 3 body.")
           if len(edge_types) != len(pts):
               raise ValueError("Počet hran musí odpovídat počtu bodů.")
           try:
               edges = tuple(Edge(EdgeType(t)) for t in edge_types)
           except ValueError as exc:
               raise ValueError(f"Neplatný typ hrany: {exc}") from exc
           if not _is_simple(pts):
               raise ValueError("Půdorys se sám protíná.")
           if abs(_signed_area(pts)) < 1e-9:
               raise ValueError("Půdorys má nulovou plochu.")
           if _signed_area(pts) < 0:  # normalizuj na CCW
               pts = list(reversed(pts))
               rev = list(reversed(edges))
               edges = tuple(rev[1:] + rev[:1])
           return cls(tuple(pts), edges)

       @property
       def n(self) -> int:
           return len(self.edges)

       def edge_vector(self, i: int) -> Point:
           a, b = self.vertices[i], self.vertices[(i + 1) % self.n]
           return (b[0] - a[0], b[1] - a[1])

       def edge_length(self, i: int) -> float:
           dx, dy = self.edge_vector(i)
           return math.hypot(dx, dy)
   ```

   ```python
   """Most: rectangle Footprint + typy hran → RoofSpec (reuse výpočtu)."""
   from __future__ import annotations

   from tectora.core.footprint import EdgeType, Footprint
   from tectora.core.spec import RoofSpec, RoofType

   _TOL = 1e-6


   class UnsupportedFootprint(ValueError):
       """Tvar, který most na RoofSpec zatím neumí (obecný polygon → skeleton)."""


   def _axis_aligned_rectangle(fp: Footprint) -> bool:
       if fp.n != 4:
           return False
       xs = sorted({round(v[0], 9) for v in fp.vertices})
       ys = sorted({round(v[1], 9) for v in fp.vertices})
       if len(xs) != 2 or len(ys) != 2:
           return False
       want = {(xs[0], ys[0]), (xs[1], ys[0]), (xs[1], ys[1]), (xs[0], ys[1])}
       return want == {(round(x, 9), round(y, 9)) for x, y in fp.vertices}


   def footprint_to_spec(fp: Footprint, pitch_deg: float, name: str = "") -> RoofSpec:
       if not _axis_aligned_rectangle(fp):
           raise UnsupportedFootprint(
               "Most umí zatím jen obdélník; obecné polygony přijdou se skeletonem.")
       xs = sorted({v[0] for v in fp.vertices})
       ys = sorted({v[1] for v in fp.vertices})
       w, d = xs[1] - xs[0], ys[1] - ys[0]
       length, width = max(w, d), min(w, d)
       eaves = [i for i, e in enumerate(fp.edges) if e.type is EdgeType.EAVE]
       n = len(eaves)

       if n == 1:
           rtype = RoofType.SHED
       elif n == 2 and (eaves[1] - eaves[0]) == 2:
           rtype = RoofType.GABLE
       elif n == 4 and abs(w - d) < _TOL:
           rtype = RoofType.PYRAMID
       elif n == 4:
           rtype = RoofType.HIP
       else:
           raise UnsupportedFootprint(
               f"Kombinace {n} okapů na obdélníku není podporovaný typ.")

       return RoofSpec(type=rtype, length=length, width=width, pitch=pitch_deg, name=name)
   ```

   ```python
   """Projekt = Footprint + globální sklon + metadata. Mapuje na JSON v1.0."""
   from __future__ import annotations

   from dataclasses import dataclass
   from typing import Optional

   from tectora.core.footprint import Footprint

   SCHEMA_VERSION = "1.0"


   @dataclass
   class Project:
       project_name: str = ""
       footprint: Optional[Footprint] = None
       global_pitch_deg: float = 35.0
       overhang_m: float = 0.0
       units: str = "m"

       def to_dict(self) -> dict:
           fp = None
           if self.footprint is not None:
               fp = {
                   "vertices": [[x, y] for x, y in self.footprint.vertices],
                   "edges": [{"type": e.type.value} for e in self.footprint.edges],
               }
           return {
               "schema_version": SCHEMA_VERSION,
               "project_name": self.project_name,
               "units": self.units,
               "footprint": fp,
               "global_pitch_deg": self.global_pitch_deg,
               "overhang_m": self.overhang_m,
           }

       @classmethod
       def from_dict(cls, data: dict) -> "Project":
           fp = None
           if data.get("footprint"):
               fp = Footprint.from_lists(
                   [(v[0], v[1]) for v in data["footprint"]["vertices"]],
                   [e["type"] for e in data["footprint"]["edges"]],
               )
           return cls(
               project_name=data.get("project_name", ""),
               footprint=fp,
               global_pitch_deg=float(data.get("global_pitch_deg", 35.0)),
               overhang_m=float(data.get("overhang_m", 0.0)),
               units=data.get("units", "m"),
           )
   ```
