# Plán implementace — Tectora Fáze 1: Headless jádro (domain → application → CLI → JSON → Excel)

> Verze 1.0 · 2026-06-23
> Předpoklad: čteš `docs/specifikace.md` (zejména §7 Architektura, §8 JSON, §9 definice).

## Cíl

Funkční, plně otestované **headless jádro** Tectory ovladatelné z příkazové řádky:
nakreslený půdorys (zadaný jako JSON / přes CLI) → výpočet výměr → výstup do JSON
i Excelu. Žádné GUI, žádné 3D — ty mají vlastní plány.

**Rozsah tohoto plánu (obdélníkové střechy):**
- pultová (1 okap), sedlová (2 protilehlé okapy + 2 štíty),
  valbová (4 okapy, obdélník), stanová (4 okapy, čtverec).
- Pokrývá akceptační případy 1, 2, 3, 5 a 6 ze specifikace §12.

**Mimo rozsah tohoto plánu (jde do navazujícího plánu):**
- Obecné polygony L/T/U + **úžlabí** (straight-skeleton) — akceptační případ 4.
- Polovalbová (vyžaduje rozšíření schématu o výšku/délku valby).
- GUI, 3D, i18n, živá synchronizace.

## Architektura (viz specifikace §7)

Hexagonální. Závislosti míří dovnitř: `cli → application → domain`,
`adapters` implementují porty z `application`. Doména závisí jen na stdlib + `numpy`.

## Tech stack

- Python ≥ 3.10, `numpy`, `openpyxl` (běh), `pytest`, `pytest-cov`, `hypothesis` (dev).
- CLI = `argparse` (stdlib, žádná nová závislost).
- Entry pointy už jsou v `pyproject.toml`: `tectora = "tectora.cli.main:main"`.

   ```python
   """Čisté geometrické pomocné funkce (bez závislosti na frameworcích)."""
   from __future__ import annotations

   import math

   Point = tuple[float, float]


   def deg_to_rad(deg: float) -> float:
       return math.radians(deg)


   def slope_factor(pitch_deg: float) -> float:
       """1 / cos(sklon) — převod půdorysné plochy na skloněnou."""
       return 1.0 / math.cos(math.radians(pitch_deg))


   def polygon_signed_area(points: list[Point]) -> float:
       """Znaménková plocha (shoelace). Kladná = proti směru hodin (CCW)."""
       n = len(points)
       acc = 0.0
       for i in range(n):
           x1, y1 = points[i]
           x2, y2 = points[(i + 1) % n]
           acc += x1 * y2 - x2 * y1
       return acc / 2.0


   def _on_segment(p: Point, q: Point, r: Point) -> bool:
       return (
           min(p[0], r[0]) - 1e-12 <= q[0] <= max(p[0], r[0]) + 1e-12
           and min(p[1], r[1]) - 1e-12 <= q[1] <= max(p[1], r[1]) + 1e-12
       )


   def _orient(a: Point, b: Point, c: Point) -> int:
       val = (b[1] - a[1]) * (c[0] - b[0]) - (b[0] - a[0]) * (c[1] - b[1])
       if abs(val) < 1e-12:
           return 0
       return 1 if val > 0 else -1


   def segments_intersect(a: Point, b: Point, c: Point, d: Point) -> bool:
       """Protínají se úsečky ab a cd (včetně dotyku)?"""
       o1, o2, o3, o4 = _orient(a, b, c), _orient(a, b, d), _orient(c, d, a), _orient(c, d, b)
       if o1 != o2 and o3 != o4:
           return True
       if o1 == 0 and _on_segment(a, c, b):
           return True
       if o2 == 0 and _on_segment(a, d, b):
           return True
       if o3 == 0 and _on_segment(c, a, d):
           return True
       if o4 == 0 and _on_segment(c, b, d):
           return True
       return False


   def is_simple_polygon(points: list[Point]) -> bool:
       """True, pokud se žádné dvě nesousedící hrany neprotínají."""
       n = len(points)
       if n < 3:
           return False
       for i in range(n):
           a, b = points[i], points[(i + 1) % n]
           for j in range(i + 1, n):
               # přeskoč sousední hrany (sdílejí vrchol)
               if j == i or (j + 1) % n == i or (i + 1) % n == j:
                   continue
               c, d = points[j], points[(j + 1) % n]
               if segments_intersect(a, b, c, d):
                   return False
       return True
   ```

4. Implementuj `src/tectora/domain/footprint.py`:
   ```python
   """Půdorys budovy: vrcholy + hrany s typem okap/štít."""
   from __future__ import annotations

   import math
   from dataclasses import dataclass
   from enum import Enum

   from . import geometry as geo
   from .errors import InvalidPolygon

   Point = tuple[float, float]


   class EdgeType(str, Enum):
       EAVE = "eave"    # okap
       GABLE = "gable"  # štít


   @dataclass(frozen=True)
   class Edge:
       type: EdgeType


   @dataclass(frozen=True)
   class Footprint:
       vertices: tuple[Point, ...]
       edges: tuple[Edge, ...]

       @classmethod
       def from_lists(cls, vertices: list[Point], edge_types: list[str]) -> "Footprint":
           pts = [(float(x), float(y)) for x, y in vertices]
           if len(pts) < 3:
               raise InvalidPolygon("Půdorys musí mít alespoň 3 body.")
           if len(edge_types) != len(pts):
               raise InvalidPolygon(
                   f"Počet hran ({len(edge_types)}) musí odpovídat počtu bodů ({len(pts)})."
               )
           try:
               edges = tuple(Edge(EdgeType(t)) for t in edge_types)
           except ValueError as exc:
               raise InvalidPolygon(f"Neplatný typ hrany: {exc}") from exc
           if not geo.is_simple_polygon(pts):
               raise InvalidPolygon("Půdorys se sám protíná.")
           if abs(geo.polygon_signed_area(pts)) < 1e-9:
               raise InvalidPolygon("Půdorys má nulovou plochu.")
           # normalizuj na CCW (kladná plocha). Po otočení bodů musí hrana i
           # stále ležet mezi vrcholy i a i+1 → new_edges[i] = old[(n-2-i) % n].
           if geo.polygon_signed_area(pts) < 0:
               pts = list(reversed(pts))
               rev = list(reversed(edges))
               edges = tuple(rev[1:] + rev[:1])
           return cls(tuple(pts), edges)

       @property
       def n(self) -> int:
           return len(self.edges)

       def edge_vector(self, i: int) -> Point:
           a = self.vertices[i]
           b = self.vertices[(i + 1) % self.n]
           return (b[0] - a[0], b[1] - a[1])

       def edge_length(self, i: int) -> float:
           dx, dy = self.edge_vector(i)
           return math.hypot(dx, dy)

       def signed_area(self) -> float:
           return geo.polygon_signed_area(list(self.vertices))
   ```

   ```python
   from __future__ import annotations

   import math
   from dataclasses import dataclass, field


   @dataclass(frozen=True)
   class RoofPlane:
       id: int
       footprint_area_m2: float   # plocha půdorysného regionu roviny
       pitch_deg: float
       azimuth_deg: float

       @property
       def area_m2(self) -> float:
           """Skutečná (skloněná) plocha."""

       planes: tuple[RoofPlane, ...]
       ridges: tuple[float, ...] = field(default_factory=tuple)   # 3D délky hřebenů
   ```

## Task 4 — Doména: generator.py (obdélníkové střechy)

> Geometrie (obdélník W×D, sklon θ, `t=tan θ`, `c=cos θ`):
> - **pultová** (1 okap): 1 rovina, plocha půdorysu = W·D; run = rozměr kolmý na okap;
>   2 kolmé hrany jsou krajové (3D = run/c); žádný hřeben/nároží/úžlabí.
> - **sedlová** (2 protilehlé okapy + 2 štíty): rozpětí S = vzdálenost mezi okapy,
>   hřeben rovnoběžný s okapy, délka = délka okapu L; 2 roviny po W·S/2 půdorysné
>   plochy; každý štít přispěje (S/2)/c do levé i pravé krajové.
> - **valbová** (4 okapy, obdélník, L=delší, B=kratší): hřeben = L−B; 4 nároží,
>   každé (B/2)·√(2+t²); 2 trojúhelníky (B²/4) + 2 lichoběžníky ((2L−B)·B/4).
> - **stanová** (4 okapy, čtverec S): hřeben = 0; 4 nároží po (S/2)·√(2+t²);
>   4 trojúhelníky po S²/4.

**Files:** `tests/domain/test_generator.py`, `src/tectora/domain/generator.py`.

   ```python
   """Generování geometrie střechy z půdorysu (obdélníkové případy)."""
   from __future__ import annotations

   import math

   from .errors import PitchOutOfRange, UnsupportedFootprint
   from .footprint import EdgeType, Footprint
   from .roof import RoofModel, RoofPlane

   _TOL = 1e-6


   def _is_axis_aligned_rectangle(fp: Footprint) -> bool:
       if fp.n != 4:
           return False
       xs = sorted({round(v[0], 9) for v in fp.vertices})
       ys = sorted({round(v[1], 9) for v in fp.vertices})
       if len(xs) != 2 or len(ys) != 2:
           return False
       want = {(xs[0], ys[0]), (xs[1], ys[0]), (xs[1], ys[1]), (xs[0], ys[1])}
       got = {(round(x, 9), round(y, 9)) for x, y in fp.vertices}
       return want == got


   def _eave_indices(fp: Footprint) -> list[int]:
       return [i for i, e in enumerate(fp.edges) if e.type is EdgeType.EAVE]


   def generate(fp: Footprint, pitch_deg: float) -> RoofModel:
       if not (1.0 <= pitch_deg <= 89.0):
           raise PitchOutOfRange(f"Sklon {pitch_deg}° je mimo rozsah 1–89°.")
       if not _is_axis_aligned_rectangle(fp):
           raise UnsupportedFootprint(
               "Tento generátor zvládá zatím jen obdélníkový půdorys "
               "(obecné polygony / úžlabí přijdou v navazujícím plánu)."
           )

       xs = sorted({v[0] for v in fp.vertices})
       ys = sorted({v[1] for v in fp.vertices})
       w = xs[1] - xs[0]   # rozměr ve směru x
       d = ys[1] - ys[0]   # rozměr ve směru y
       c = math.cos(math.radians(pitch_deg))
       t = math.tan(math.radians(pitch_deg))

       eaves = _eave_indices(fp)
       n_eaves = len(eaves)

       if n_eaves == 1:
           return _shed(fp, w, d, c, eaves[0], pitch_deg)
       # dvě protilehlé hrany 4úhelníku se liší indexem o 2
       if n_eaves == 2 and (eaves[1] - eaves[0]) == 2:
           return _gable(fp, w, d, c, eaves, pitch_deg)
       if n_eaves == 4 and abs(w - d) < _TOL:
           return _tent(w, c, t, pitch_deg)
       if n_eaves == 4:
           return _hip(w, d, c, t, pitch_deg)

       raise UnsupportedFootprint(
           f"Kombinace {n_eaves} okapů na obdélníku není podporovaný typ střechy."
       )


   def _shed(fp: Footprint, w: float, d: float, c: float, eave_idx: int, pitch: float) -> RoofModel:
       # orientaci okapu urči z geometrie hrany, ne z parity indexu
       dx, dy = fp.edge_vector(eave_idx)
       horizontal_eave = abs(dx) >= abs(dy)
       eave_len = w if horizontal_eave else d
       run = d if horizontal_eave else w
       rake = run / c
       plane = RoofPlane(1, w * d, pitch, 0.0)
       return RoofModel(
           planes=(plane,),
           eaves=(eave_len,),
           rakes_left=(rake,),
           rakes_right=(rake,),
       )


   def _gable(fp: Footprint, w: float, d: float, c: float, eaves: list[int], pitch: float) -> RoofModel:
       dx, dy = fp.edge_vector(eaves[0])
       horizontal_eaves = abs(dx) >= abs(dy)
       eave_len = w if horizontal_eaves else d   # délka okapu = délka hřebene
       span = d if horizontal_eaves else w       # rozpětí mezi okapy
       half = span / 2.0
       plane_fp = eave_len * half
       planes = (
           RoofPlane(1, plane_fp, pitch, 0.0),
           RoofPlane(2, plane_fp, pitch, 180.0),
       )
       rake = half / c
       return RoofModel(
           planes=planes,
           ridges=(eave_len,),
           eaves=(eave_len, eave_len),
           rakes_left=(rake, rake),    # po jedné krajové na každý ze 2 štítů
           rakes_right=(rake, rake),
       )


   def _hip(w: float, d: float, c: float, t: float, pitch: float) -> RoofModel:
       L = max(w, d)
       B = min(w, d)
       ridge = L - B
       hip = (B / 2.0) * math.sqrt(2.0 + t * t)
       tri_fp = B * B / 4.0
       trap_fp = (2.0 * L - B) * B / 4.0
       planes = (
           RoofPlane(1, trap_fp, pitch, 0.0),
           RoofPlane(2, trap_fp, pitch, 180.0),
           RoofPlane(3, tri_fp, pitch, 90.0),
           RoofPlane(4, tri_fp, pitch, 270.0),
       )
       return RoofModel(
           planes=planes,
           ridges=(ridge,),
           hips=(hip, hip, hip, hip),
           eaves=(w, d, w, d),
       )


   def _tent(s: float, c: float, t: float, pitch: float) -> RoofModel:
       hip = (s / 2.0) * math.sqrt(2.0 + t * t)
       tri_fp = s * s / 4.0
       planes = tuple(
           RoofPlane(i + 1, tri_fp, pitch, 90.0 * i) for i in range(4)
       )
       return RoofModel(
           planes=planes,
           hips=(hip, hip, hip, hip),
           eaves=(s, s, s, s),
       )
   ```

   ```python
   """Agregát projektu = footprint + globální sklon + metadata. Mapuje na JSON v1.0."""
   from __future__ import annotations

   from dataclasses import dataclass
   from typing import Optional

   from tectora.domain.footprint import Footprint

   SCHEMA_VERSION = "1.0"


   @dataclass
   class Project:
       project_name: str = ""
       footprint: Optional[Footprint] = None
       global_pitch_deg: float = 35.0
       overhang_m: float = 0.0
       units: str = "m"

       def to_dict(self) -> dict:
           fp_dict = None
           if self.footprint is not None:
               fp_dict = {
                   "vertices": [[x, y] for x, y in self.footprint.vertices],
                   "edges": [{"type": e.type.value} for e in self.footprint.edges],
               }
           return {
               "schema_version": SCHEMA_VERSION,
               "project_name": self.project_name,
               "units": self.units,
               "footprint": fp_dict,
               "global_pitch_deg": self.global_pitch_deg,
               "overhang_m": self.overhang_m,
           }

       @classmethod
       def from_dict(cls, data: dict) -> "Project":
           fp = None
           fp_data = data.get("footprint")
           if fp_data:
               fp = Footprint.from_lists(
                   [(v[0], v[1]) for v in fp_data["vertices"]],
                   [e["type"] for e in fp_data["edges"]],
               )
           return cls(
               project_name=data.get("project_name", ""),
               footprint=fp,
               global_pitch_deg=float(data.get("global_pitch_deg", 35.0)),
               overhang_m=float(data.get("overhang_m", 0.0)),
               units=data.get("units", "m"),
           )
   ```

   ```python
   from __future__ import annotations

   import dataclasses

   from tectora.application.project import Project
   from tectora.application.ports import ReportExporter

   def set_footprint(project: Project, vertices: list, edge_types: list[str]) -> Project:
       project.footprint = Footprint.from_lists(vertices, edge_types)
       return project


   def set_edge_type(project: Project, index: int, edge_type: str) -> Project:
       fp = project.footprint

       new_edges[index] = Edge(EdgeType(edge_type))
   ```

   ```python
   """Adaptér: export výkazu výměr do Excelu (.xlsx)."""
   from __future__ import annotations

   from openpyxl import Workbook

   from tectora.domain.measurements import summarize
   from tectora.domain.roof import RoofModel


   class ExcelReportExporter:
       def export(self, roof: RoofModel, path: str) -> None:
           m = summarize(roof)
           wb = Workbook()
           ws = wb.active
           ws.title = "Výkaz"
           rows = [
               ("Veličina", "Hodnota"),
               ("Plocha střechy (m²)", round(m.area_m2, 3)),
               ("Délka hřebene (m)", round(m.ridge_length_m, 3)),
               ("Délka nároží (m)", round(m.hip_length_m, 3)),
               ("Počet nároží", m.hip_count),
               ("Délka úžlabí (m)", round(m.valley_length_m, 3)),
               ("Počet úžlabí", m.valley_count),
               ("Obvod okapu (m)", round(m.eave_perimeter_m, 3)),
               ("Krajová hrana levá (m)", round(m.rake_left_m, 3)),
               ("Krajová hrana pravá (m)", round(m.rake_right_m, 3)),
           ]
           for row in rows:
               ws.append(row)

           ws2 = wb.create_sheet("Roviny")
           ws2.append(("ID", "Plocha (m²)", "Sklon (°)", "Azimut (°)"))
           for p in roof.planes:
               ws2.append((p.id, round(p.area_m2, 3), p.pitch_deg, p.azimuth_deg))

           wb.save(path)
   ```

   ```python
   """CLI adaptér — řídí ho člověk i AI agent (Claude Code)."""
   from __future__ import annotations

   import argparse
   import json
   import sys

   from tectora.adapters.excel_exporter import ExcelReportExporter
   from tectora.adapters.json_repository import JsonProjectRepository
   from tectora.application import use_cases as uc
   from tectora.application.dto import result_to_dict
   from tectora.application.project import Project
   from tectora.domain.errors import TectoraError

   _repo = JsonProjectRepository()


   def _parse_vertices(text: str) -> list[tuple[float, float]]:
       points = []
       for part in text.split(";"):
           part = part.strip()
           if not part:
               continue
           x, y = part.split(",")
           points.append((float(x), float(y)))
       return points


   def _emit_error(message: str) -> int:
       json.dump({"error": {"message": message}}, sys.stdout, ensure_ascii=False)
       sys.stdout.write("\n")
       return 1


   def _build_parser() -> argparse.ArgumentParser:
       parser = argparse.ArgumentParser(prog="tectora", description="Kalkulátor střech")
       sub = parser.add_subparsers(dest="command", required=True)

       p_new = sub.add_parser("new", help="Vytvoří nový projekt")
       p_new.add_argument("path")
       p_new.add_argument("--name", default="")

       p_fp = sub.add_parser("set-footprint", help="Nastaví půdorys")
       p_fp.add_argument("path")
       p_fp.add_argument("--vertices", required=True,
                         help='Body "x,y; x,y; ..." v metrech')

       p_edge = sub.add_parser("set-edge", help="Nastaví typ hrany")
       p_edge.add_argument("path")
       p_edge.add_argument("--index", type=int, required=True)
       p_edge.add_argument("--type", choices=["eave", "gable"], required=True)

       p_pitch = sub.add_parser("set-pitch", help="Nastaví globální sklon")
       p_pitch.add_argument("path")
       p_pitch.add_argument("--deg", type=float, required=True)

       p_calc = sub.add_parser("calculate", help="Spočítá výměry")
       p_calc.add_argument("path")
       p_calc.add_argument("--output", help="Zapíše výsledek do JSON")

       p_exp = sub.add_parser("export", help="Export výkazu do Excelu")
       p_exp.add_argument("path")
       p_exp.add_argument("--xlsx", required=True)

       p_show = sub.add_parser("show", help="Čitelné shrnutí")
       p_show.add_argument("path")
       return parser


   def main(argv: list[str] | None = None) -> int:
       args = _build_parser().parse_args(argv)
       try:
           if args.command == "new":
               _repo.save(args.path, uc.create_project(args.name))
               return 0
           if args.command == "set-footprint":
               project = _repo.load(args.path)
               verts = _parse_vertices(args.vertices)
               types = ["eave"] * len(verts)
               project = uc.set_footprint(project, verts, types)
               _repo.save(args.path, project)
               return 0
           if args.command == "set-edge":
               project = _repo.load(args.path)
               project = uc.set_edge_type(project, args.index, args.type)
               _repo.save(args.path, project)
               return 0
           if args.command == "set-pitch":
               project = _repo.load(args.path)
               project = uc.set_pitch(project, args.deg)
               _repo.save(args.path, project)
               return 0
           if args.command == "calculate":
               project = _repo.load(args.path)
               roof = uc.calculate_roof(project)
               result = result_to_dict(roof)
               if args.output:
                   with open(args.output, "w", encoding="utf-8") as fh:
                       json.dump(result, fh, ensure_ascii=False, indent=2)
               else:
                   json.dump(result, sys.stdout, ensure_ascii=False, indent=2)
                   sys.stdout.write("\n")
               return 0
           if args.command == "export":
               project = _repo.load(args.path)
               roof = uc.calculate_roof(project)
               uc.export_report(roof, ExcelReportExporter(), args.xlsx)
               return 0
           if args.command == "show":
               project = _repo.load(args.path)
               roof = uc.calculate_roof(project)
               result = result_to_dict(roof)
               t = result["totals"]
               print(f"Plocha střechy: {t['area_m2']:.2f} m²")
               print(f"Hřeben: {t['ridge_length_m']:.2f} m")
               print(f"Nároží: {t['hip_count']} ks / {t['hip_length_m']:.2f} m")
               print(f"Úžlabí: {t['valley_count']} ks / {t['valley_length_m']:.2f} m")
               print(f"Okap: {t['eave_perimeter_m']:.2f} m")
               print(f"Krajová L/P: {t['rake_left_m']:.2f} / {t['rake_right_m']:.2f} m")
               return 0
       except (TectoraError, ValueError) as exc:
           return _emit_error(str(exc))
       return 0


   if __name__ == "__main__":
       raise SystemExit(main())
   ```
